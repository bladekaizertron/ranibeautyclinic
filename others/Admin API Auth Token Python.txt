import base64
import hashlib
import hmac
import time

def generate_auth_header(business_id, api_secret, api_key):
    prefix = 'blvd-admin-v1'
    timestamp = str(int(time.time()))

    payload = f'{prefix}{business_id}{timestamp}'.encode('utf-8')
    raw_key = base64.b64decode(api_secret)
    signature = hmac.new(raw_key, payload, hashlib.sha256).digest()
    signature_base64 = base64.b64encode(signature).decode('utf-8')
    token = f'{signature_base64}{payload.decode("utf-8")}'
    http_basic_payload = f'{api_key}:{token}'
    http_basic_credentials = base64.b64encode(http_basic_payload.encode('utf-8')).decode('utf-8')

    return http_basic_credentials

BUSINESS_ID = 'your_business_id'
API_SECRET = 'your_api_secret'
API_KEY = 'your_api_key'

print(generate_auth_header(BUSINESS_ID, API_SECRET, API_KEY))
